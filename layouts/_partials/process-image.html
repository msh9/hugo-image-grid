($src.Process (printf "resize x%d avif" $height))) }} {{ $jpg = merge $jpg (dict (string .) ($src.Process (printf "resize x%d jpg" $height))) }} {{ end }} {{ $result = dict "original" . "alt" $alt "avif" $avif "jpg" $jpg }} {{ else }} {{ warnf "hugo-image-grid: skipping %q, it is not an image Hugo can process" .RelPermalink }} {{ end }} {{ return $result }}
